後端白名單驗證邏輯設定 (GAS 環境)

// ALLOWED_SALES_IDS 白名單清單

const ALLOWED_SALES_IDS = [

'U_SALES_01', // 林士傑

'U_SALES_02', // 陳雅芳

'U_DIRECTOR' // 業務主管

];


function verifySalesAccess(userId) {

if (!ALLOWED_SALES_IDS.includes(userId)) {

throw new Error('未授權存取！非合格業務人員。');

}

return true;

}

透過上述驗證機制，即使 LINE Bot Webhook 接收到任何未授權 ID 的詢價訊息，後端均第一時間攔截並拒絕回傳底價或客戶資料。
